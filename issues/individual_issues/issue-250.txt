def points_to_pixels(

    pixtocams,
    camtoworlds,
    distortion_params = None,
    camtype = ProjectionType.PERSPECTIVE,
    xnp = np,
):

  """
  This function calculates the 2D pixel coordinates and depth values from 3D point coordinates using camera intrinsics, extrinsics, and optionally distortion parameters.   Input-Output Arguments
  :param points: float array, 3D coordinates of points to project. These are the world coordinates that need to be projected onto the 2D image plane.

  :param distortion_params: dict of floats or float arrays, optional. These parameters are used to model the camera lens distortion. If provided, the function will correct for radial and tangential distortion.
  :param camtype: camera_utils.ProjectionType, specifies the type of camera projection model to use. Currently, only perspective projection is supported.
  :param xnp: module, either numpy or jax.numpy. This parameter allows the function to be used for computations on different devices (CPU or GPU/TPU).
  :return: A tuple containing two elements:
           coordinates: float array, the computed 2D pixel coordinates of the input 3D points.
           depth: float array, the depth values of the points in the camera coordinate system, useful for depth-aware applications.
  """
