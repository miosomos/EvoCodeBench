def pos_enc(x, min_deg, max_deg, append_identity=True):

  """
  This function generates a positional encoding for a given input, based on the method used in the original NeRF paper. It scales the input, applies a sine function, and optionally appends the original input to the result.

  Input-Output Arguments
  :param x: Array. The input array to be encoded.
  :param min_deg: Integer. The minimum degree exponent for scaling the input.
  :param max_deg: Integer. The maximum degree exponent for scaling the input. The scales are generated as powers of 2 from min_deg to max_deg.
  :param append_identity: Bool, optional. Determines whether the original input array should be concatenated with the encoded array. Defaults to True.
  :return: Array. The resulting array after applying positional encoding. If append_identity is True, the original input is concatenated with the encoded features.
  """
